MULTISELECTOPTION INTERFACE
---------------------------
Standard option interface:

interface MultiSelectOption {
  value: string | number;         // Unique identifier
  label: string;                  // Display text
  icon?: string;                  // Icon or emoji
  subtitle?: string;              // Secondary text
  group?: string;                 // Group name
  disabled?: boolean;             // Disabled state
  selected?: boolean;             // Initial selection
}

Usage:
const options: MultiSelectOption[] = [
  { value: 1, label: 'JavaScript', icon: '🟨' },
  { value: 2, label: 'TypeScript', icon: '🔷' }
];

GENERIC TYPE SUPPORT
--------------------
Use generics for custom objects. Every callback takes ONE ctx; per-option callbacks read
ctx.option:

interface User {
  id: number;
  firstName: string;
  lastName: string;
  email: string;
}

const users: User[] = [...];
const select = document.querySelector('web-multiselect') as MultiSelectElement<User>;

// Type-safe callbacks
select.getValueCallback = ({ option }: ItemContext<User>): number => option.id;
select.getDisplayValueCallback = ({ option }: ItemContext<User>): string =>
  `${option.firstName} ${option.lastName}`;

Or extend MultiSelectOption:
interface UserOption extends Partial<MultiSelectOption> {
  id: number;
  firstName: string;
  lastName: string;
  email: string;
}

CONTEXT BASE
------------
Every callback ctx and every event detail extends MultiSelectContext:

interface MultiSelectContext<T = any> {
  controller: MultiSelectController<T>;   // non-enumerable
  element: HTMLElement;                    // the host, non-enumerable
  presentation: 'floating' | 'fullscreen';
  isFullscreen: boolean;
  isModal: boolean;
}

EVENT DETAIL TYPES
------------------
Event detail types (one per event):

interface SelectEventDetail<T = any> extends MultiSelectContext<T> {
  option: T;                           // Selected option
  selectedOptions: T[];                // All selected options
  selectedValues: (string | number)[];
}

type DeselectEventDetail<T = any> = SelectEventDetail<T>;   // option = the deselected one

interface ChangeEventDetail<T = any> extends MultiSelectContext<T> {
  selectedOptions: T[];                // All selected options
  selectedValues: (string | number)[]; // Values only
}

interface AddEventDetail<T = any> extends MultiSelectContext<T> {
  value: string;                       // Typed text
  option?: T;                          // Created option (when beforeAddCallback returned one)
  selectedOptions: T[];
  selectedValues: (string | number)[];
}

type ReadyEventDetail<T = any> = MultiSelectContext<T>;

type MultiSelectActionEventDetail<T = any>  // ActionContext + action, id, item, source,
                                            // sourceEvent, triggerElement, dataset

Usage:
select.addEventListener('change', (e: CustomEvent<ChangeEventDetail>) => {
  console.log(e.detail.selectedOptions);
  console.log(e.detail.selectedValues);
});

// on* twins receive the DETAIL (not the CustomEvent):
select.onChange = (detail: ChangeEventDetail<User>) => console.log(detail.selectedValues);
select.onSelect = ({ option }: SelectEventDetail<User>) => console.log(option.email);

Action button type (core action item):

interface MultiSelectActionItem<T = any> {
  action?: 'select-all' | 'clear-selection';   // built-in; omit for a custom item
  id?: string;                           // identifies a custom item
  text?: string;                         // plain text
  icon?: string;                         // glyph / emoji (text)
  tooltip?: string;
  cssClass?: string;
  row?: number;
  isVisible?: boolean;
  isDisabled?: boolean;
  isDanger?: boolean;
  isToggled?: boolean;
  getTextCallback?(ctx: ActionContext<T>): string;
  getIconCallback?(ctx: ActionContext<T>): string;
  getTooltipCallback?(ctx: ActionContext<T>): string;
  getClassCallback?(ctx: ActionContext<T>): string | string[];
  getIsVisibleCallback?(ctx: ActionContext<T>): boolean;
  getIsDisabledCallback?(ctx: ActionContext<T>): boolean;
  getIsToggledCallback?(ctx: ActionContext<T>): boolean;
  onClick?(detail: MultiSelectActionEventDetail<T>): void | Promise<void>;
}

interface ActionContext<T = any> extends MultiSelectContext<T> {
  selectedValues: (string | number)[];
  selectedOptions: T[];
  options: ReadonlyArray<T>;
  selectedCount: number;
  optionCount: number;
  isOpen: boolean;
  searchTerm: string;
  option?: T;      // for data-action triggers in option / badge content
}

Usage:
const actions: Array<MultiSelectActionItem | 'select-all' | 'clear-selection'> = [
  'select-all',
  {
    id: 'export',
    text: 'Export',
    icon: '⬇️',
    getIsDisabledCallback: ({ selectedCount }) => selectedCount === 0,
    onClick: ({ selectedOptions }) => exportItems(selectedOptions)
  }

select.actionButtons = actions;

GENERIC COMPONENT TYPE
----------------------
Type-safe component reference:

function setupMultiSelect<T>(
  selector: string,
  options: T[]
): MultiSelectElement<T> {
  const select = document.querySelector(selector) as MultiSelectElement<T>;

  select.options = options;

  select.onChange = ({ selectedOptions }: ChangeEventDetail<T>) => {
    console.log('Selected:', selectedOptions);
  };

  return select;
}

// Usage
interface Product {
  id: number;
  name: string;
  price: number;
}

const select = setupMultiSelect<Product>('web-multiselect', products);

TYPE GUARDS
-----------
Check types at runtime:

function isMultiSelectOption(obj: any): obj is MultiSelectOption {
  return (
    obj &&
    typeof obj === 'object' &&
    ('value' in obj) &&
    ('label' in obj)
  );
}

if (isMultiSelectOption(data)) {
  // TypeScript knows data is MultiSelectOption
  console.log(data.value, data.label);
}

FRAMEWORK INTEGRATION
---------------------
React with TypeScript:

import { useRef, useEffect } from 'react';
import type { MultiSelectElement, MultiSelectOption, ChangeEventDetail } from '@keenmate/web-multiselect';

interface Props {
  options: MultiSelectOption[];
  onChange: (selected: MultiSelectOption[]) => void;
}

function MyComponent({ options, onChange }: Props) {
  const selectRef = useRef<MultiSelectElement>(null);

  useEffect(() => {
    const select = selectRef.current;
    if (!select) return;

    select.options = options;

    const handleChange = (e: Event) => {
      const detail = (e as CustomEvent<ChangeEventDetail<MultiSelectOption>>).detail;
      onChange(detail.selectedOptions);
    };

    select.addEventListener('change', handleChange);

    return () => {
      select.removeEventListener('change', handleChange);
    };
  }, [options, onChange]);

  return <web-multiselect ref={selectRef} />;
}

Vue 3 with TypeScript:

<script setup lang="ts">
import { ref, onMounted } from 'vue';
import type { MultiSelectElement, MultiSelectOption } from '@keenmate/web-multiselect';

interface Props {
  options: MultiSelectOption[];
}

const props = defineProps<Props>();
const select = ref<MultiSelectElement | null>(null);

onMounted(() => {
  if (select.value) {
    select.value.options = props.options;
  }
});
</script>

<template>
  <web-multiselect ref="select"></web-multiselect>
</template>

STRICT NULL CHECKS
------------------
Handle nullable types:

function getSelectValue(select: MultiSelectElement | null): (string | number)[] {
  if (!select) return [];

  const value = select.selectedValue;

  if (Array.isArray(value)) {
    return value;
  }

  return [value];
}
